// Global State ve Supabase Bağlantısı const S = { user: null, profile: null, routines: [] }; // Kendi Supabase bilgilerini buraya yaz const SUPABASE_URL = 'BURAYA_SUPABASE_URL_YAZ'; const SUPABASE_ANON_KEY = 'BURAYA_SUPABASE_ANON_KEY_YAZ'; const sb = window.supabase.createClient(SUPABASE_URL, SUPABASE_ANON_KEY); // 1. Uygulama Açılış Fonksiyonu async function boot() { try { const { data: { session } } = await sb.auth.getSession(); if (session && session.user) { S.user = session.user; await loadUserProfile(); renderApp(); } else { renderLogin(); } } catch (err) { console.error("Boot hatası:", err); renderLogin(); } } // 2. Profil Bilgilerini Veritabanından Çekme async function loadUserProfile() { if (!S.user) return; try { const { data, error } = await sb .from('profiles') .select('*') .eq('id', S.user.id) .single(); if (data) { S.profile = data; } else { console.warn("Profil bulunamadı:", error); } } catch (err) { console.error("Profil yüklenirken hata:", err); } } // 3. Giriş ve Kayıt İşlemlerini Yöneten Ana Fonksiyon async function handleAuthAction(email, password, username = '', displayName = '', isRegister = false) { if (isRegister) { // Kayıt Olma Adımı const { data, error } = await sb.auth.signUp({ email, password }); if (error) { alert('Kayıt hatası: ' + error.message); return; } if (data.user) { // profiles tablosuna kullanıcı adı ve görünen adı kaydet const { error: profileErr } = await sb.from('profiles').insert([ { id: data.user.id, username: username, display_name: displayName, photo: null } ]); if (profileErr) { console.error('Profil kaydedilemedi:', profileErr.message); alert('Kayıt olundu ancak profil detayları eklenirken hata oluştu: ' + profileErr.message); } else { // social tablosuna başlangıç kaydı at await sb.from('social').insert([{ username: username, following: [], followers: [] }]); alert('Kayıt başarılı! Lütfen e-postanıza gelen onay linkine tıklayın.'); } } } else { // Giriş Yapma Adımı const { data, error } = await sb.auth.signInWithPassword({ email, password }); if (error) { alert('Giriş başarısız: ' + error.message); return; } S.user = data.user; await loadUserProfile(); renderApp(); } } // 4. Çıkış Yapma Fonksiyonu async function handleLogout() { await sb.auth.signOut(); S.user = null; S.profile = null; renderLogin(); } // 5. Arayüz Çizim Fonksiyonları (Örnek Yapı - Kendi HTML yapına göre özelleştirebilirsin) function renderLogin() { const app = document.getElementById('app') || document.body; app.innerHTML = `

Ritim Giriş / Kayıt

`; } function renderApp() { const app = document.getElementById('app') || document.body; const displayName = S.profile ? S.profile.display_name : (S.user ? S.user.email : 'Kullanıcı'); const username = S.profile ? S.profile.username : ''; app.innerHTML = `

Hoş Geldin, ${displayName}

Kullanıcı Adı: @${username}

E-posta: ${S.user ? S.user.email : ''}

`; } // 6. Buton Olay Bağlantıları window.onLoginClick = function() { const email = document.getElementById('email').value; const pass = document.getElementById('password').value; handleAuthAction(email, pass, '', '', false); }; window.onRegisterClick = function() { const email = document.getElementById('email').value; const pass = document.getElementById('password').value; const username = document.getElementById('username').value; const displayName = document.getElementById('displayName').value; handleAuthAction(email, pass, username, displayName, true); }; // Sayfa Yüklendiğinde Başlat window.addEventListener('DOMContentLoaded', () => { boot(); });
