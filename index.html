// Global Durum (State) Objesi const S = { user: null, profile: null, routines: [] }; // Supabase İstemcisi (Kendi projene göre anahtarların burada olmalı) const SUPABASE_URL = 'SENIN_SUPABASE_URL'; const SUPABASE_ANON_KEY = 'SENIN_SUPABASE_ANON_KEY'; const sb = window.supabase.createClient(SUPABASE_URL, SUPABASE_ANON_KEY); // Uygulamayı Başlatan Ana Fonksiyon async function boot(){ try { const { data: { session } } = await sb.auth.getSession(); if(session && session.user){ S.user = session.user; await loadUserProfile(); render(); } else { renderLogin(); } } catch (err) { console.error("Boot hatası:", err); renderLogin(); } } // Kullanıcı Profilini ve Detaylarını Veritabanından Çekme async function loadUserProfile() { if (!S.user) return; try { const { data, error } = await sb .from('profiles') .select('*') .eq('id', S.user.id) .single(); if (data) { S.profile = data; } else { console.warn("Profil bulunamadı:", error); } } catch (err) { console.error("Profil yüklenirken hata:", err); } } // Giriş ve Kayıt İşlemlerini Yöneten Fonksiyon async function handleAuthAction(email, password, username = '', displayName = '', isRegister = false) { if (isRegister) { // 1. Supabase Auth ile Kayıt Ol const { data, error } = await sb.auth.signUp({ email, password }); if (error) { alert('Kayıt hatası: ' + error.message); return; } if (data.user) { // 2. profiles tablosuna kullanıcı adını ve görünen adı kaydet const { error: profileErr } = await sb.from('profiles').insert([ { id: data.user.id, username: username, display_name: displayName, photo: null } ]); if (profileErr) { console.error('Profil kaydedilemedi:', profileErr.message); alert('Kayıt olundu ancak profil detayları eklenirken hata oluştu: ' + profileErr.message); } else { // 3. social tablosuna başlangıç kaydı at await sb.from('social').insert([{ username: username, following: [], followers: [] }]); alert('Kayıt başarılı! Lütfen e-postanıza gelen onay linkine tıklayın.'); } } } else { // Giriş Yapma İşlemi const { data, error } = await sb.auth.signInWithPassword({ email, password }); if (error) { alert('Giriş başarısız: ' + error.message); return; } S.user = data.user; await loadUserProfile(); render(); } } // Arayüz Render Fonksiyonları (Örnek Yapı) function renderLogin() { const app = document.getElementById('app') || document.body; app.innerHTML = `

Ritim Giriş / Kayıt

`; } function render() { const app = document.getElementById('app') || document.body; const displayName = S.profile ? S.profile.display_name : (S.user ? S.user.email : 'Kullanıcı'); const username = S.profile ? S.profile.username : ''; app.innerHTML = `

Hoş Geldin, ${displayName}

Kullanıcı Adı: @${username}

E-posta: ${S.user ? S.user.email : ''}

`; } // Buton Olayları Yardımcıları window.handleLoginClick = function() { const email = document.getElementById('email').value; const pass = document.getElementById('password').value; handleAuthAction(email, pass, '', '', false); }; window.handleRegisterClick = function() { const email = document.getElementById('email').value; const pass = document.getElementById('password').value; const username = document.getElementById('username').value; const displayName = document.getElementById('displayName').value; handleAuthAction(email, pass, username, displayName, true); }; window.handleLogout = async function() { await sb.auth.signOut(); S.user = null; S.profile = null; renderLogin(); }; // Sayfa Yüklendiğinde Başlat window.addEventListener('DOMContentLoaded', () => { boot(); });
